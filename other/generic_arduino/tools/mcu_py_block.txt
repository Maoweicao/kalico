        if self.is_fileoutput():
            return {"is_config": 0, "move_count": 500, "crc": 0}
        config_params = get_config_cmd.send()
        # Trust the MCU's atomic config response for is_shutdown state.
        # Do NOT check self._is_shutdown here — a transient shutdown
        # message may arrive between the config response and this check
        # (common on generic_arduino during init), and the MCU's own
        # config handler has already told us the authoritative answer.
        if config_params["is_shutdown"]:
            raise error(
                "Can not update MCU '%s' config as it is shutdown"
                % (self._name,)
            )
        return config_params
